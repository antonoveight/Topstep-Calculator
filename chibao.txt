f_patterns(float c1o, float c1h, float c1l, float c1c, float c2o, float c2h, float c2l, float c2c) =>
    float lowRef = tsReference == "C1 range" ? c1l : manualSsl > 0 ? manualSsl : na
    float highRef = tsReference == "C1 range" ? c1h : manualBsl > 0 ? manualBsl : na
    bool tookLow = not na(lowRef) and c2l <= lowRef - sweepSize
    bool tookHigh = not na(highRef) and c2h >= highRef + sweepSize
    bool reclaimed = c2c > c1l and c2c < c1h
    bool sweptBothC1 = c2l <= c1l - sweepSize and c2h >= c1h + sweepSize
    bool doubleSweep = sweptBothC1 or (tookLow and tookHigh)
    bool tsBull = tookLow and not doubleSweep and reclaimed and c2c > lowRef
    bool tsBear = tookHigh and not doubleSweep and reclaimed and c2c < highRef
    [tsBull, tsBear]

type BiasState
    int direction = 0
    int age = 0
    float swingHigh = na
    float swingLow = na
    float protectedHigh = na
    float protectedLow = na

// Only confirmed pivots and closes in the bias timeframe reach this state machine.
f_biasStep(BiasState state, float ph, float pl, float closing, float priorClose) =>
    state.age += 1
    if not na(ph)
        state.swingHigh := ph
    if not na(pl)
        state.swingLow := pl
    bool brokeHigh = not na(state.swingHigh) and closing > state.swingHigh and priorClose <= state.swingHigh
    bool brokeLow = not na(state.swingLow) and closing < state.swingLow and priorClose >= state.swingLow
    bool lostBull = state.direction == 1 and not na(state.protectedLow) and closing < state.protectedLow
    bool lostBear = state.direction == -1 and not na(state.protectedHigh) and closing > state.protectedHigh
    int nextDirection = brokeHigh or lostBear ? 1 : brokeLow or lostBull ? -1 : 0
    if nextDirection != 0 and not (brokeHigh and brokeLow)
        if state.direction != nextDirection
            state.age := 0
        state.direction := nextDirection
        state.protectedLow := nextDirection == 1 ? state.swingLow : na
        state.protectedHigh := nextDirection == -1 ? state.swingHigh : na
    // Trail only with confirmed structural swings; a single opposing candle is insufficient.
    if state.direction == 1 and not na(pl)
        state.protectedLow := na(state.protectedLow) ? pl : math.max(state.protectedLow, pl)
    if state.direction == -1 and not na(ph)
        state.protectedHigh := na(state.protectedHigh) ? ph : math.min(state.protectedHigh, ph)

f_biasCore() =>
    var BiasState state = BiasState.new()
    float ph = ta.pivothigh(high, htfPivotLength, htfPivotLength)
    float pl = ta.pivotlow(low, htfPivotLength, htfPivotLength)
    f_biasStep(state, ph, pl, close, close[1])
    float guardHigh = state.direction == -1 and not na(state.protectedHigh) ? math.min(high, state.protectedHigh) : high
    float guardLow = state.direction == 1 and not na(state.protectedLow) ? math.max(low, state.protectedLow) : low
    [state.direction, state.age, state.swingHigh, state.swingLow, guardHigh, guardLow]
f_biasSnapshot(bool previous) =>
    [direction, age, swingHigh, swingLow, guardHigh, guardLow] = f_biasCore()
    [previous ? direction[1] : direction, previous ? age[1] : age, previous ? swingHigh[1] : swingHigh, previous ? swingLow[1] : swingLow, previous ? guardHigh[1] : guardHigh, previous ? guardLow[1] : guardLow]

// Context outcomes, not executed trade outcomes. OHLC cannot order both barriers.
f_crtEnd(int direction, float invalidation, float targetLevel, float barHigh, float barLow, float closing, bool older, bool outsideHistory, bool timedOut, bool entryExpired = false) =>
    bool bad = older and (direction == 1 ? barLow <= invalidation : barHigh >= invalidation)
    bool reached = older and (direction == 1 ? barHigh >= targetLevel : barLow <= targetLevel)
    bad and reached ? 4 : bad ? 1 : reached ? 2 : entryExpired ? 7 : outsideHistory ? 5 : timedOut ? 3 : 0

// References are matched BEFORE current-bar pool consumption/creation.
f_keyReact(CrtKey k, int beforeTime, int direction, float barHigh, float barLow, float barClose) =>
    bool preexisting = k.kind > 0 and not na(k.knownAt) and k.knownAt <= beforeTime
    bool sameSide = k.dir == direction
    bool sweepReaction = direction == 1 ? barLow <= k.lo - sweepSize and barClose > k.hi : barHigh >= k.hi + sweepSize and barClose < k.lo
    bool enabled = keyTypes == "All" or keyTypes == "Swing" and k.kind == 3 or keyTypes == "Swing + FVG" and (k.kind == 3 or k.kind == 1) or keyTypes == "Swing + OB" and (k.kind == 3 or k.kind == 2) or keyTypes == "Open / Close" and (k.kind == 4 or k.kind == 5) or keyTypes == "Weekly wick" and k.kind == 6
    bool zoneReaction = barHigh >= k.lo and barLow <= k.hi and (direction == 1 ? barClose > k.hi : barClose < k.lo)
    preexisting and sameSide and enabled and (k.kind == 3 ? sweepReaction : zoneReaction)

f_findCrtKey(array<Liquidity> swings, int beforeTime, int direction, float barHigh, float barLow, float barClose, string sourceTf) =>
    CrtKey found = CrtKey.new()
    if array.size(swings) > 0
        for offset = 0 to array.size(swings) - 1
            Liquidity q = array.get(swings, array.size(swings) - 1 - offset)
            CrtKey candidate = CrtKey.new(kind=3, dir=-q.side, hi=q.price, lo=q.price, born=q.born, leftTime=q.born, knownAt=q.knownAt, beforeTime=beforeTime, tf=sourceTf)
            if found.kind == 0 and f_keyReact(candidate, beforeTime, direction, barHigh, barLow, barClose)
                found := candidate
    found

f_keyIdentity(CrtKey key) =>
    key.tf + ":" + str.tostring(key.kind) + ":" + str.tostring(key.dir) + ":" + str.tostring(key.knownAt)

f_matchKeyMap(KeyMap keyMap, array<string> usedKeys, int beforeTime, int direction, float barHigh, float barLow, float barClose) =>
    CrtKey found = CrtKey.new()
    for pass = 1 to 6
        if array.size(keyMap.levels) > 0
            for j = 0 to array.size(keyMap.levels) - 1
                CrtKey candidate = array.get(keyMap.levels, j)
                if found.kind == 0 and candidate.kind == pass and not array.includes(usedKeys, f_keyIdentity(candidate)) and f_keyReact(candidate, beforeTime, direction, barHigh, barLow, barClose)
                    found := candidate.copy()
                    found.beforeTime := beforeTime
    found

f_markHigherUsed(KeyMap keyMap, array<string> usedKeys, int barStart, float barHigh, float barLow) =>
    if array.size(keyMap.levels) > 0
        for j = 0 to array.size(keyMap.levels) - 1
            CrtKey key = array.get(keyMap.levels, j)
            bool consumed = key.kind == 3 and (key.dir == 1 ? barLow <= key.lo : barHigh >= key.hi)
            string identity = f_keyIdentity(key)
            if barStart >= key.knownAt and consumed and not array.includes(usedKeys, identity)
                array.push(usedKeys, identity)
    while array.size(usedKeys) > 160
        array.shift(usedKeys)

f_updateKeyPools(array<Liquidity> swings, int scanStart, bool inScan, float ph, float pl) =>
    // Remove consumed or out-of-window references, before looking for new CP levels.
    int si = array.size(swings) - 1
    while si >= 0
        Liquidity q = array.get(swings, si)
        if q.born < scanStart or (q.side == 1 ? high >= q.price : low <= q.price)
            array.remove(swings, si)
        si -= 1
    if inScan and not na(ph) and time[htfPivotLength] >= scanStart and high < ph
        array.push(swings, Liquidity.new(knownAt=time_close, price=ph, side=1, born=time[htfPivotLength]))
    if inScan and not na(pl) and time[htfPivotLength] >= scanStart and low > pl
        array.push(swings, Liquidity.new(knownAt=time_close, price=pl, side=-1, born=time[htfPivotLength]))
    if array.size(swings) > 80
        array.shift(swings)
f_collectKeyMap(array<Liquidity> swings, string sourceTf) =>
    array<CrtKey> levels = array.new<CrtKey>()
    for direction = -1 to 1 by 2
        int collected = 0
        if array.size(swings) > 0
            for offset = 0 to array.size(swings) - 1
                Liquidity q = array.get(swings, array.size(swings) - 1 - offset)
                if -q.side == direction and collected < 2
                    array.push(levels, CrtKey.new(kind=3, dir=direction, hi=q.price, lo=q.price, born=q.born, leftTime=q.born, knownAt=q.knownAt, tf=sourceTf))
                    collected += 1
    KeyMap.new(levels)

// Key provenance is HTF-only. All newly created keys become usable NEXT bar.
// FVG = wick gap of three closed candles; OB = last opposite candle (<=5)
// before a close breaks its full wick and the preceding bar, not an LTF entry OB.
f_pushKey(array<CrtKey> zones, int kind, int direction, float hi, float lo, int left, string sourceTf) =>
    array.push(zones, CrtKey.new(kind=kind, dir=direction, hi=hi, lo=lo, born=left, leftTime=left, knownAt=time_close, tf=sourceTf))
    while array.size(zones) > 40
        array.shift(zones)
f_zoneKeys(array<CrtKey> zones, string sourceTf) =>
    int i = array.size(zones) - 1
    while i >= 0
        CrtKey k = array.get(zones, i)
        bool broken = k.dir == 1 ? close < k.lo : close > k.hi
        bool old = time - k.knownAt > int(timeframe.in_seconds() * 1000) * 48
        if broken or old
            array.remove(zones, i)
        i -= 1
    if not na(high[2]) and low > high[2]
        f_pushKey(zones, 1, 1, low, high[2], time[2], sourceTf)
    if not na(low[2]) and high < low[2]
        f_pushKey(zones, 1, -1, low[2], high, time[2], sourceTf)
    bool up = close > open and close > high[1]
    bool down = close < open and close < low[1]
    if up or down
        bool found = false
        for back = 1 to 5
            bool opposite = up ? close[back] < open[back] : close[back] > open[back]
            bool displaced = up ? close > high[back] and close[1] <= high[back] : close < low[back] and close[1] >= low[back]
            if not found and opposite
                found := true
                if displaced
                    int direction = up ? 1 : -1
                    f_pushKey(zones, 2, direction, high[back], low[back], time[back], sourceTf)
                    f_pushKey(zones, 4, direction, open[back], open[back], time[back], sourceTf)
                    f_pushKey(zones, 5, direction, close[back], close[back], time[back], sourceTf)
    zones
// Mechanical personal reference: midpoint of each real wick, not candle EQ.
// Created in W1 and exposed only through its previous completed snapshot.
f_weeklyWickMap(float opening, float barHigh, float barLow, float closing, int openedAt, int closedAt) =>
    array<CrtKey> levels = array.new<CrtKey>()
    float bodyLow = math.min(opening, closing)
    float bodyHigh = math.max(opening, closing)
    if not na(closedAt) and bodyLow - barLow > sweepSize
        float lowerMid = (barLow + bodyLow) / 2
        array.push(levels, CrtKey.new(kind=6, dir=1, lo=lowerMid, hi=lowerMid, born=openedAt, leftTime=openedAt, knownAt=closedAt, tf="1W"))
    if not na(closedAt) and barHigh - bodyHigh > sweepSize
        float upperMid = (barHigh + bodyHigh) / 2
        array.push(levels, CrtKey.new(kind=6, dir=-1, lo=upperMid, hi=upperMid, born=openedAt, leftTime=openedAt, knownAt=closedAt, tf="1W"))
    KeyMap result = KeyMap.new(levels)
    result.sourceClose := closing
    result.sourceEnd := closedAt
    result.sourceTf := "1W"
    result
f_weeklyWickSnapshot() =>
    KeyMap result = f_weeklyWickMap(open, high, low, close, time, time_close)
    result[1]

f_higherKeyMap(string sourceTf) =>
    var array<Liquidity> swings = array.new<Liquidity>()
    var array<CrtKey> zones = array.new<CrtKey>()
    int scanStart = last_bar_time - int(timeframe.in_seconds() * 1000) * scanDepth
    float ph = ta.pivothigh(high, htfPivotLength, htfPivotLength)
    float pl = ta.pivotlow(low, htfPivotLength, htfPivotLength)
    f_updateKeyPools(swings, scanStart, time >= scanStart, ph, pl)
    f_zoneKeys(zones, sourceTf)
    KeyMap result = f_collectKeyMap(swings, sourceTf)
    result.sourceClose := close
    result.sourceEnd := time_close
    result.sourceTf := sourceTf
    // Most recent two of each zone type/direction; bounded snapshot size 20.
    for kind = 1 to 5
        if kind != 3
            for direction = -1 to 1 by 2
                int kept = 0
                if array.size(zones) > 0
                    for back = 0 to array.size(zones) - 1
                        CrtKey k = array.get(zones, array.size(zones) - 1 - back)
                        if kept < 2 and k.kind == kind and k.dir == direction
                            array.push(result.levels, k.copy())
                            kept += 1
    result

f_biasBundle() =>
    [direction, age, swingHigh, swingLow, guardHigh, guardLow] = f_biasSnapshot(true)
    KeyMap keyMap = f_higherKeyMap(biasTf)
    [direction, age, swingHigh, swingLow, guardHigh, guardLow, keyMap[1]]

// Freeze the references that existed before C1. They are evidence for this parent,
// not a claim that liquidity can be traded repeatedly without a new raid/reclaim.
f_parentKeys(array<Liquidity> swings, KeyMap higherMap, array<string> consumed, int beforeTime, string sourceTf) =>
    array<CrtKey> levels = array.new<CrtKey>()
    if array.size(swings) > 0
        for i = 0 to array.size(swings) - 1
            Liquidity q = array.get(swings, i)
            if q.knownAt <= beforeTime
                array.push(levels, CrtKey.new(kind=3, dir=-q.side, hi=q.price, lo=q.price, born=q.born, leftTime=q.born, knownAt=q.knownAt, beforeTime=beforeTime, tf=sourceTf))
    if not na(higherMap) and not na(higherMap.levels)
        if array.size(higherMap.levels) > 0
            for i = 0 to array.size(higherMap.levels) - 1
                CrtKey k = array.get(higherMap.levels, i)
                if k.knownAt <= beforeTime and not array.includes(consumed, f_keyIdentity(k))
                    CrtKey copy = k.copy()
                    copy.beforeTime := beforeTime
                    array.push(levels, copy)
    KeyMap.new(levels)

f_rangeReaction(KeyMap references, int beforeTime, int direction, float raidHigh, float raidLow, float closing) =>
    CrtKey found = CrtKey.new()
    if not na(references) and not na(references.levels)
        if array.size(references.levels) > 0
            for i = 0 to array.size(references.levels) - 1
                CrtKey k = array.get(references.levels, i)
                if found.kind == 0 and f_keyReact(k, beforeTime, direction, raidHigh, raidLow, closing)
                    found := k.copy()
    found

// Execution keys are current, completed HTF references. A model's historical
// key is context; it never proves that a NEW LTF P3 has reacted at a key.
f_executionSnapshot(string tf) =>
    KeyMap keys = f_higherKeyMap(tf)
    KeyMap previous = keys[1]
    KeyMap result = KeyMap.new(array.new<CrtKey>())
    if not na(previous)
        result.sourceClose := previous.sourceClose
        result.sourceEnd := previous.sourceEnd
        result.sourceTf := previous.sourceTf
        for k in previous.levels
            array.push(result.levels, k.copy())
    // W1 already applies its own [1]; append AFTER the local offset to avoid
    // an extra H1/H4 delay at the week boundary (or a second weekly offset).
    KeyMap weekly = request.security(syminfo.tickerid, "1W", f_weeklyWickSnapshot(), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
    if not na(weekly) and not na(weekly.levels)
        for k in weekly.levels
            array.push(result.levels, k.copy())
    result
KeyMap executionH1 = request.security(syminfo.tickerid, "60", f_executionSnapshot("60"), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
KeyMap executionMain = request.security(syminfo.tickerid, htf, f_executionSnapshot(htf), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
KeyMap executionLarger = request.security(syminfo.tickerid, largerTf, f_executionSnapshot(largerTf), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
KeyMap executionBias = request.security(syminfo.tickerid, biasTf, f_executionSnapshot(biasTf), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)

f_executionKey(int direction, int knownBefore, float peakHigh, float peakLow, float reclaimClose) =>
    CrtKey found = f_rangeReaction(executionH1, knownBefore, direction, peakHigh, peakLow, reclaimClose)
    if found.kind == 0
        found := f_rangeReaction(executionMain, knownBefore, direction, peakHigh, peakLow, reclaimClose)
    if found.kind == 0
        found := f_rangeReaction(executionLarger, knownBefore, direction, peakHigh, peakLow, reclaimClose)
    if found.kind == 0
        found := f_rangeReaction(executionBias, knownBefore, direction, peakHigh, peakLow, reclaimClose)
    if found.kind > 0
        found.beforeTime := knownBefore
    found

// Freeze the nearest opposing HTF key known at the local TS close.
// This is an intermediate objective; it does not replace the CRT range target.
f_nextBarrier(KeyMap references, int beforeTime, int direction, float origin, float targetLevel) =>
    CrtKey found = CrtKey.new()
    float best = na
    if not na(references) and not na(references.levels)
        for k in references.levels
            float level = direction == 1 ? k.lo : k.hi
            float distance = direction == 1 ? level - origin : origin - level
            float room = direction == 1 ? targetLevel - origin : origin - targetLevel
            bool enabled = keyTypes == "All" or keyTypes == "Swing" and k.kind == 3 or keyTypes == "Swing + FVG" and (k.kind == 3 or k.kind == 1) or keyTypes == "Swing + OB" and (k.kind == 3 or k.kind == 2) or keyTypes == "Open / Close" and (k.kind == 4 or k.kind == 5) or keyTypes == "Weekly wick" and k.kind == 6
            bool eligible = enabled and k.kind > 0 and k.dir == -direction and not na(k.knownAt) and k.knownAt <= beforeTime and distance > sweepSize and distance < room
            if eligible and (na(best) or distance < best)
                found := k.copy()
                found.beforeTime := beforeTime
                best := distance
    found

f_forwardKey(int direction, int beforeTime, float origin, float targetLevel) =>
    CrtKey found = f_nextBarrier(executionH1, beforeTime, direction, origin, targetLevel)
    CrtKey nextexecutionMain = f_nextBarrier(executionMain, beforeTime, direction, origin, targetLevel)
    if nextexecutionMain.kind > 0 and (found.kind == 0 or (direction == 1 ? nextexecutionMain.lo < found.lo : nextexecutionMain.hi > found.hi))
        found := nextexecutionMain
    CrtKey nextexecutionLarger = f_nextBarrier(executionLarger, beforeTime, direction, origin, targetLevel)
    if nextexecutionLarger.kind > 0 and (found.kind == 0 or (direction == 1 ? nextexecutionLarger.lo < found.lo : nextexecutionLarger.hi > found.hi))
        found := nextexecutionLarger
    CrtKey nextexecutionBias = f_nextBarrier(executionBias, beforeTime, direction, origin, targetLevel)
    if nextexecutionBias.kind > 0 and (found.kind == 0 or (direction == 1 ? nextexecutionBias.lo < found.lo : nextexecutionBias.hi > found.hi))
        found := nextexecutionBias
    found

f_keyProofValid(CrtKey key, int reactedAt, float sourceClose, int sourceEnd) =>
    bool known = not na(key) and key.kind > 0 and not na(reactedAt) and not na(key.knownAt) and key.knownAt <= key.beforeTime
    bool subsequentClose = not na(sourceEnd) and sourceEnd > reactedAt
    bool broken = known and subsequentClose and (key.dir == 1 ? sourceClose < key.lo : sourceClose > key.hi)
    known and not broken

f_executionProofValid(CrtKey key, int reactedAt) =>
    // Weekly levels stay fixed; subsequent CLOSED H1 prices validate local reclaim.
    KeyMap source = not na(key) and (key.kind == 6 or key.tf == "60") ? executionH1 : not na(key) and key.tf == htf ? executionMain : not na(key) and key.tf == largerTf ? executionLarger : not na(key) and key.tf == biasTf ? executionBias : na
    not na(source) and f_keyProofValid(key, reactedAt, source.sourceClose, source.sourceEnd)

// Price-based attempt failure is sticky. Revalidation below creates a new attempt,
// with a new confirmation token, and never rewrites the previous archive or SL.
f_revalidateRange(Model m, float opening, float barHigh, float barLow, float closing, int barTime, int barEnd, int currentIndex) =>
    // Only the ORIGINAL C3 may replace the original C2 raid. Never pair a stale
    // C1 with an unrelated raid hours/days later or chain renewals indefinitely.
    bool originalC3 = m.flags == 1 and currentIndex == m.bornIndex + 1 and barTime >= m.bEnd
    bool extended = m.dir == 1 ? barLow <= m.inv - sweepSize : barHigh >= m.inv + sweepSize
    bool inside = closing > m.lo and closing < m.hi
    CrtKey key = originalC3 and extended and inside ? f_rangeReaction(m.rangeKeys, m.bT, m.dir, barHigh, barLow, closing) : CrtKey.new()
    bool pathOpen = m.dir == 1 ? barHigh < m.cp : barLow > m.cp
    bool renewed = m.phase == 0 and originalC3 and extended and inside and pathOpen and key.kind > 0
    if renewed
        m.flags += 2
        m.phase := 1
        m.bornIndex := currentIndex
        m.confirmedAt := barEnd
        m.resolvedAt := barEnd
        m.resolvedIndex := currentIndex
        m.inv := m.dir == 1 ? barLow : barHigh
        m.invT := barTime
        m.crtKey := key
        m.bO := opening
        m.bH := barHigh
        m.bL := barLow
        m.bC := closing
        m.bT := barTime
        m.bEnd := barEnd
        m.cT := na
        m.cEnd := na
        m.cO := na
        m.cH := na
        m.cL := na
        m.cC := na
    renewed

f_preferCrt(int candidateBorn, int candidateRoot, int selectedBorn, int selectedRoot, int currentIndex) =>
    na(selectedRoot) or (candidateBorn == currentIndex and selectedBorn != currentIndex) or ((candidateBorn == currentIndex) == (selectedBorn == currentIndex) and candidateRoot > selectedRoot)

f_engine(string sourceTf) =>
    [higherBias, higherBiasAge, higherBiasHigh, higherBiasLow, higherGuardHigh, higherGuardLow, higherKeyMap] = request.security(syminfo.tickerid, biasTf, f_biasBundle(), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
    var array<Model> models = array.new<Model>()
    var array<Liquidity> swings = array.new<Liquidity>()
    var ModelArchive lastEnded = ModelArchive.new()
    var array<string> usedHigherKeys = array.new<string>()
    var int priorFocusId = na
    bool archivedThisBar = false
    var int lastRetired = na
    var int retiredReason = 0
    var int dropped = 0
    var int firstSeen = time
    int scanStart = last_bar_time - int(timeframe.in_seconds() * 1000) * scanDepth
    bool inScan = time >= scanStart
    float ph = ta.pivothigh(high, htfPivotLength, htfPivotLength)
    float pl = ta.pivotlow(low, htfPivotLength, htfPivotLength)
    KeyMap localMap = f_higherKeyMap(sourceTf)
    KeyMap priorLocalMap = localMap[1]
    KeyMap parentReferences = f_parentKeys(swings, higherKeyMap, usedHigherKeys, time, sourceTf)
    KeyMap weeklyReferences = request.security(syminfo.tickerid, "1W", f_weeklyWickSnapshot(), lookahead=barmerge.lookahead_on, calc_bars_count=CONTEXT_BARS)
    if not na(weeklyReferences) and not na(weeklyReferences.levels)
        for k in weeklyReferences.levels
            CrtKey frozenWeekly = k.copy()
            frozenWeekly.beforeTime := time
            array.push(parentReferences.levels, frozenWeekly)
    if not na(priorLocalMap) and not na(priorLocalMap.levels)
        for k in priorLocalMap.levels
            CrtKey frozen = k.copy()
            frozen.beforeTime := time
            array.push(parentReferences.levels, frozen)
    CrtKey priorBuyKey = f_rangeReaction(parentReferences, time, 1, high, low, close)
    CrtKey priorSellKey = f_rangeReaction(parentReferences, time, -1, high, low, close)
    if biasTf != sourceTf and not na(higherKeyMap) and not na(higherKeyMap.levels)
        if priorBuyKey.kind == 0
            priorBuyKey := f_matchKeyMap(higherKeyMap, usedHigherKeys, time, 1, high, low, close)
        if priorSellKey.kind == 0
            priorSellKey := f_matchKeyMap(higherKeyMap, usedHigherKeys, time, -1, high, low, close)
        f_markHigherUsed(higherKeyMap, usedHigherKeys, time, high, low)
    f_updateKeyPools(swings, scanStart, inScan, ph, pl)

    // A parent C1 outlives any individual attempt. Goal completion ends the parent.
    int mi = array.size(models) - 1
    while mi >= 0
        Model m = array.get(models, mi)
        bool older = time_close > m.confirmedAt
        if na(m.cT) and time >= m.confirmedAt and m.dir != 0
            m.cO := open
            m.cH := high
            m.cL := low
            m.cC := close
            m.cT := time
            m.cEnd := time_close
        float equilibrium = (m.hi + m.lo) / 2
        if older and (m.dir == 1 ? high >= equilibrium : low <= equilibrium)
            m.eqTouched := true
        int endingReason = m.phase == 1 ? f_crtEnd(m.dir, m.inv, m.cp, high, low, close, older, false, false) : 0
        bool goalTaken = older and (m.dir == 1 ? high >= m.cp : low <= m.cp)
        bool outsideHistory = m.aT < scanStart
        if endingReason != 0
            lastRetired := m.uid
            retiredReason := endingReason
            if f_archiveAllowed(m.kind) and (m.uid == priorFocusId or not archivedThisBar)
                lastEnded := f_archive(m, endingReason, time_close)
                archivedThisBar := true
            m.phase := 0
        if goalTaken or outsideHistory
            lastRetired := m.uid
            retiredReason := goalTaken ? 2 : 5
            array.remove(models, mi)
        else
            // A wick beyond the old SL may become a NEW raid; it never repairs the old attempt.
            bool renewed = older and f_revalidateRange(m, open, high, low, close, time, time_close, bar_index)
            // A failed attempt without a valid immediate C3 replacement is over.
            // Do not retain it as a parent waiting for an unrelated future raid.
            if m.phase == 0
                array.remove(models, mi)
        mi -= 1

    [tsBull, tsBear] = f_patterns(open[1], high[1], low[1], close[1], open, high, low, close)
    bool crt = tsBull or tsBear
    if inScan and not na(time[1]) and crt
        // With timeout disabled, old context-only records must not starve a new C3.
        bool freshPath = tsBull ? high < high[1] : low > low[1]
        if array.size(models) >= queueLimit and freshPath
            Model oldest = array.shift(models)
            lastRetired := oldest.uid
            retiredReason := 6
            if oldest.uid == priorFocusId or not archivedThisBar
                lastEnded := f_archive(oldest, 6, time_close)
                archivedThisBar := true
        if array.size(models) < queueLimit
            int direction = tsBull ? 1 : -1
            float rh = high[1]
            float rl = low[1]
            Model fresh = Model.new(rangeKeys=parentReferences, bEnd=time_close, aEnd=time_close[1], uid=time[1], flags=1, kind=1, phase=1, dir=direction, bornIndex=bar_index, confirmedAt=time_close, resolvedAt=time_close, hi=rh, lo=rl, inv=direction == 1 ? low : high, cp=direction == 1 ? rh : rl, aO=open[1], aH=high[1], aL=low[1], aC=close[1], aT=time[1], bO=open, bH=high, bL=low, bC=close, bT=time)
            fresh.crtKey := direction == 1 ? priorBuyKey : priorSellKey
            fresh.hiT := time[1]
            fresh.loT := time[1]
            fresh.invT := time
            fresh.cpT := time[1]
            bool targetAtBirth = direction == 1 ? high >= fresh.cp : low <= fresh.cp
            if targetAtBirth
                lastRetired := fresh.uid
                retiredReason := 4
                if not archivedThisBar and f_archiveAllowed(fresh.kind)
                    lastEnded := f_archive(fresh, 4, time_close)
                    archivedThisBar := true
            else
                array.push(models, fresh)
        else
            dropped += 1



    // Entry-open CRT first; otherwise the nearest still-valid C1 for tracking.
    Model selected = Model.new()
    if array.size(models) > 0
        for i = 0 to array.size(models) - 1
            Model candidate = array.get(models, i)
            if candidate.phase == 1 and f_preferCrt(candidate.bornIndex, candidate.aT, selected.bornIndex, selected.aT, bar_index)
                selected := candidate
    priorFocusId := selected.uid
    bool historyReady = firstSeen <= scanStart
    [selected.uid, selected.flags, selected.kind, selected.phase, selected.dir, selected.hi, selected.lo, selected.inv, selected.cp, selected.erlU, selected.erlL, selected.irlH, selected.irlL, selected.irlT, selected.irlKind, selected.confirmedAt, selected.resolvedAt, array.size(models), selected.aO, selected.aH, selected.aL, selected.aC, selected.aT, selected.bO, selected.bH, selected.bL, selected.bC, selected.bT, selected.cO, selected.cH, selected.cL, selected.cC, selected.cT, lastRetired, retiredReason, dropped, historyReady, bar_index - selected.bornIndex, selected.hiT, selected.loT, selected.invT, selected.cpT, selected.erlUT, selected.erlLT, selected.bullIrlH, selected.bullIrlL, selected.bullIrlT, selected.bullIrlKind, selected.bearIrlH, selected.bearIrlL, selected.bearIrlT, selected.bearIrlKind, selected.bullIrlLeft, selected.bearIrlLeft, selected.innerHi, selected.innerLo, selected.innerHiT, selected.innerLoT, lastEnded, selected.crtKey, selected.eqTouched, selected.aEnd, selected.cEnd, selected.bEnd]

    int endedNow = 0

        state.reason := 0
        state.endedAt := na

// MPS confirms only the OTE branch, never the direct TS or FVG branch.
f_stepChannel(SignalContext ctx, ReactionState state, float knownHigh, float knownLow, float pivotHigh, float pivotLow, bool waveAllowed = true, bool forcePullback = false, bool mpsAllowed = true, int knownHighAt = na, int knownLowAt = na) =>

    result.status := not isLtf ? "VIEW TF" : not armed ? "WAIT CRT" : "WAIT CLOSE"
    if armed and barstate.isconfirmed and isLtf and time >= state.activatedAt and (na(state.lastClosedAt) or state.lastClosedAt != time_close)
        state.lastClosedAt := time_close

            state.sourceRoot := ctx.c1T
            state.sourceC2 := ctx.c2T
            state.sourceEnd := ctx.c2End
        bool chainAllowed = na(ctx.c1T) or (state.sourceRoot == ctx.c1T and state.sourceC2 == ctx.c2T and state.sourceEnd == ctx.c2End )
        bool bullish = ctx.dir == 1

        bool targetBar = bullish ? high >= ctx.cp : low <= ctx.cp

        state.localTargetTaken := state.localTargetTaken or targetBar
        bool ended = state.localStopTaken or state.localTargetTaken
        bool priorKey = not na(ctx.crtKey) and ctx.crtKey.kind > 0 and ctx.crtKey.dir == ctx.dir and not na(ctx.crtKey.knownAt) and ctx.crtKey.knownAt <= ctx.crtKey.beforeTime and ctx.crtKey.beforeTime < ctx.confirmed

        // predecessor can still reclaim P1; that predecessor is not the CRT range.
        bool orderedPair = not na(knownHigh) and not na(knownLow) and knownHigh > knownLow and not na(knownHighAt) and not na(knownLowAt) and (bullish ? knownLowAt < knownHighAt and knownHighAt < time : knownHighAt < knownLowAt and knownLowAt < time)

            state.mpsRefAt := na

            state.p3KeyBroken := false

            state.confirmTime := na
            state.mpsAt := na
            state.gapAt := na
        // An unreclaimed candidate can advance to a newer completed P1/P2 pair.
        // After the FIRST reclaim, freeze that parent through further raid wicks.

            state.p3KeyBroken := false

        bool localTs = structuralTs or (not hasRaid and adjacentTs)
        // A fresh TS starts its own branch budget. No global direction latch.
        bool newTs = localTs and not ended and (na(state.confirmIndex) or state.branchMask == 7)
        if newTs
            state.forwardKey := f_forwardKey(ctx.dir, time, close, ctx.cp)
            state.nextObjective := state.forwardKey.kind > 0 ? (bullish ? state.forwardKey.lo : state.forwardKey.hi) : na
            state.objectiveReached := false
            state.objectiveAt := na
            state.confirmIndex := bar_index
            state.confirmTime := time
            state.confirmLevel := hasRaid ? state.p1Ref : bullish ? low[1] : high[1]
            state.localExtreme := hasRaid ? state.raidExtreme : bullish ? low : high
            bool retainedPeakKey = hasRaid and not na(state.p3Key) and f_keyReact(state.p3Key, state.raidAt, ctx.dir, state.p3High, state.p3Low, close)
            state.p3Key := retainedPeakKey ? state.p3Key : f_executionKey(ctx.dir, hasRaid ? state.raidAt : time, hasRaid ? state.p3High : high, hasRaid ? state.p3Low : low, close)
            state.p3KeyBroken := false
            state.p3KeyAt := state.p3Key.kind > 0 ? time_close : na
            state.branchMask := 0
            if hasRaid
                state.raidReclaimed := true
            else
                state.mpsRef := na
                state.mpsRefAt := na
            state.mpsAt := na
            state.legExtreme := bullish ? high : low
            state.gapAt := na
            state.gapOrigin := na
            state.gapBroken := false
            state.sent := false
        bool hasTs = not na(state.confirmIndex)
        chainAllowed := chainAllowed and (na(ctx.c2T) or na(state.p3At) or state.p3At >= ctx.c2T)
        state.chainRole := not chainAllowed ? "SOURCE LOST" : na(ctx.c2End) or na(state.p3At) ? "LOCAL" : state.p3At < ctx.c2End ? "FROM C2" : not na(ctx.c3End) and state.p3At < ctx.c3End ? "LOCAL C3" : "LOCAL TRACK"
        bool later = hasTs and bar_index > state.confirmIndex
        // A nearby key is an observation, not the CRT target or a trade exit.
        // Count touches only AFTER the TS close; OHLC cannot order its own wick.
        bool reachedForward = later and not na(state.nextObjective) and (bullish ? high >= state.nextObjective : low <= state.nextObjective)
        if reachedForward and not state.objectiveReached
            state.objectiveReached := true
            state.objectiveAt := time_close
        if hasTs and not na(state.p3KeyAt) and not f_executionProofValid(state.p3Key, state.p3KeyAt)
            state.p3KeyBroken := true
        bool localKeyGood = not state.p3KeyBroken and not na(state.p3Key) and state.p3Key.dir == ctx.dir and f_executionProofValid(state.p3Key, state.p3KeyAt)
        bool keyAllowed = not requireKey or localKeyGood
        bool counterStructure = not na(state.p1At) and not na(state.mpsRefAt) and not na(state.p3At) and state.p1At < state.mpsRefAt and state.mpsRefAt < state.p3At and state.p3At <= state.confirmTime and (bullish ? state.localExtreme <= state.p1Ref - sweepSize and close > state.p1Ref : state.localExtreme >= state.p1Ref + sweepSize and close < state.p1Ref)
        bool counterContext = allowCrtCounter and opposingBias and keyHeld and localKeyGood and counterStructure
        bool parentKnown = ctx.parentLive and ctx.parentAtKey and ctx.parentDir != 0 and not na(ctx.parentConfirmed) and ctx.parentConfirmed <= time
        bool againstParent = parentKnown and ctx.parentDir != ctx.dir
        bool parentReversal = allowCrtCounter and keyHeld and localKeyGood and counterStructure
        bool contextAllowed = not againstParent or parentReversal
        bool biasAllowed = (not filterBias or biasDir == ctx.dir or counterContext) and contextAllowed
        state.biasRole := not biasAllowed ? (biasDir == 0 ? "B0" : againstParent ? "HTF CONFLICT" : "BIAS") : againstParent ? "COUNTER SWING" : counterContext ? (parentKnown ? "HTF KEY REV" : "KEY REV") : parentKnown and ctx.confirmed >= ctx.parentConfirmed and localKeyGood ? "CONT" : "FLOW"
        // The source extreme must follow P2, including evidence inherited from
        // Early C2. A whole-C2 high/low from BEFORE P2 cannot become its P3.
        bool mpsOrder = not na(state.p1At) and not na(state.mpsRefAt) and not na(state.p3At) and state.p1At < state.mpsRefAt and state.mpsRefAt < state.p3At and state.p3At <= state.confirmTime
        bool mpsSweep = not na(state.p1Ref) and not na(state.localExtreme) and (bullish ? state.localExtreme <= state.p1Ref - sweepSize : state.localExtreme >= state.p1Ref + sweepSize)
        bool mpsNow = mpsAllowed and structureBarReady and hasTs and chainAllowed and mpsOrder and mpsSweep and localKeyGood and na(state.mpsAt) and not na(state.mpsRef) and (bullish ? structureClosing > state.mpsRef and structurePriorClose <= state.mpsRef : structureClosing < state.mpsRef and structurePriorClose >= state.mpsRef)
        if mpsNow and not ended
            state.mpsAt := structureClosedAt
            state.mpsEvent := true
        // OTE is calculated from the PREVIOUSLY KNOWN first-leg endpoint.
        // A bar extending the endpoint cannot also retrospectively use that zone.
        float legBefore = state.legExtreme
        float span = hasTs ? math.abs(legBefore - state.localExtreme) : na
        float oteLow = bullish ? legBefore - span * 0.8 : legBefore + span * 0.5
        float oteHigh = bullish ? legBefore - span * 0.5 : legBefore + span * 0.8
        bool extendsLeg = hasTs and (bullish ? high > legBefore : low < legBefore)
        bool ote = later and not na(state.mpsAt) and time >= state.mpsAt and span > sweepSize and not extendsLeg and low <= oteHigh and high >= oteLow and oppositeClose and (bullish ? close > oteLow : close < oteHigh)
        // Test an EXISTING gap before discovering a new one on this close.
        bool gapReady = later and not na(state.gapAt) and time >= state.gapAt and not state.gapBroken
        bool gapInvalid = gapReady and (bullish ? close < state.gapLow : close > state.gapHigh)
        bool fvg = gapReady and not gapInvalid and low <= state.gapHigh and high >= state.gapLow and oppositeClose and (bullish ? close > state.gapHigh : close < state.gapLow)
        if gapInvalid
            state.gapBroken := true
        bool canEmit = not ended and chainAllowed and keyAllowed and biasAllowed and timeAllowed and waveAllowed and (signalCadence != "One per model" or not state.everSent)
        bool direct = newTs and entryBranches != "FVG / OTE"
        bool fvgEntry = fvg and entryBranches != "TS close" and not f_has(state.branchMask, 2)
        bool oteEntry = ote and entryBranches != "TS close" and not f_has(state.branchMask, 4)
        int emitted = (direct ? 1 : 0) + (fvgEntry ? 2 : 0) + (oteEntry ? 4 : 0)
        result.tsDetected := localTs
        result.retestDetected := fvg or ote
        result.legDetected := state.mpsEvent
        result.entryDetected := emitted > 0
        result.status := ended ? (state.localStopTaken ? "INV" : "TGT") : not hasTs ? "WAIT TS" : not keyAllowed ? "WAIT P3 KEY" : not biasAllowed ? ((opposingBias or againstParent) and not counterStructure ? "WAIT CT SWING" : "BIAS") : signalCadence == "One per model" and state.everSent ? "USED" : not na(state.mpsAt) ? "WAIT OTE/FVG" : not localKeyGood ? "MPS NO KEY" : "WAIT FVG/MPS"
        if canEmit and emitted > 0
            result.hit := true
            result.tag := (direct ? "TS_CLOSE" : "") + (fvgEntry ? (direct ? "+FVG" : "TS_FVG") : "") + (oteEntry ? (direct or fvgEntry ? "+OTE" : "MPS_OTE") : "")
            result.confirmTime := state.confirmTime
            result.counterBias := counterContext
            result.keyRef := state.p3Key
            result.status := "SIGNAL"
            result.zoneLow := fvgEntry ? state.gapLow : oteEntry ? oteLow : na
            result.zoneHigh := fvgEntry ? state.gapHigh : oteEntry ? oteHigh : na
            result.zoneKnownAt := fvgEntry ? state.gapAt : oteEntry ? time : na
            result.referencePrice := state.mpsRef
            result.legKnownAt := state.mpsAt
            state.branchMask += emitted
            state.sent := true
            state.everSent := true
            state.lastSignalTime := time
            state.lastSignalBar := bar_index
            state.lastSignalHigh := high
            state.lastSignalLow := low
        // Freeze the first leg once the OTE branch is consumed.
        if hasTs and not f_has(state.branchMask, 4)
            state.legExtreme := bullish ? math.max(legBefore, high) : math.min(legBefore, low)
        bool gapSource = later and time[1] >= state.confirmTime
        bool newGap = gapSource and (bullish ? low > high[2] : high < low[2])
        if newGap and not f_has(state.branchMask, 2) and na(state.gapAt)
            state.gapLow := bullish ? high[2] : high
            state.gapHigh := bullish ? low : low[2]
            state.gapAt := time_close
            state.gapOrigin := time[2]
            state.gapBroken := false
    result

type SmtBars

f_smtBars() =>
    SmtBars.new(first=time[2], firstEnd=time_close[2], second=time[1], secondEnd=time_close[1], firstHigh=high[2], firstLow=low[2], secondHigh=high[1], secondLow=low[1])
f_smtDirection(SmtBars own, SmtBars peer, bool inverse) =>
    bool synchronized = not na(own.first) and not na(peer.first) and own.first == peer.first and own.second == peer.second and own.firstEnd == peer.firstEnd and own.secondEnd == peer.secondEnd
    bool ownHigh = own.secondHigh > own.firstHigh
    bool ownLow = own.secondLow < own.firstLow
    bool peerHigh = peer.secondHigh > peer.firstHigh
    bool peerLow = peer.secondLow < peer.firstLow
    bool buy = synchronized and ownLow and not (inverse ? peerHigh : peerLow)
    bool sell = synchronized and ownHigh and not (inverse ? peerLow : peerHigh)
    buy and not sell ? 1 : sell and not buy ? -1 : 0

f_stepEarly(EarlyReference ref, EarlyState state, CrtKey buyKey, CrtKey sellKey, bool buyWave, bool sellWave, bool confirmedHit, bool permitEntry = true, int parentDirection = 0, bool qualifiedParent = false, int parentKnownAt = na) =>
    EarlyResult result = EarlyResult.new(bucket=ref.start, c1Time=ref.c1Time)
    if isLtf and barstate.isconfirmed and not na(ref.c1Time) and time >= ref.start
        bool newWindow = na(state.bucket) or state.bucket != ref.start
        if newWindow
            state.root := ref.c1Time
            state.bucket := ref.start
            state.lastClose := ref.start
            state.dir := 0
            state.hi := high
            state.lo := low
            state.hiAt := time
            state.loAt := time
            state.highCandleLow := low
            state.lowCandleHigh := high
            state.setupP1 := na
            state.setupP1At := na
            state.setupP2 := na
            state.setupP2At := na
            state.complete := time == ref.start
            state.blocked := false
            state.emitted := false
            state.attempts := 0
            state.reclaimedAt := na
            state.attemptSl := na
            state.lockedSl := na
            state.lockedTp := na
            state.key := CrtKey.new()
            state.channel := ReactionState.new()
            state.terminalReason := ""
        if not newWindow and time != state.lastClose
            state.complete := false
        state.lastClose := time_close
        bool extended = state.dir == 1 ? low < state.lo : state.dir == -1 ? high > state.hi : false
        // Keep the latest actual touch when an equal extreme is revisited.
        // The companion wick must belong to that same candle.
        if high >= state.hi
            state.hiAt := time
            state.highCandleLow := low
        if low <= state.lo
            state.loAt := time
            state.lowCandleHigh := high
        state.hi := math.max(state.hi, high)
        state.lo := math.min(state.lo, low)
        bool raidLow = state.lo <= ref.c1L - sweepSize
        bool raidHigh = state.hi >= ref.c1H + sweepSize
        bool both = raidLow and raidHigh
        bool inside = close > ref.c1L and close < ref.c1H
        if state.dir == 0 and not both
            state.dir := raidLow ? 1 : raidHigh ? -1 : 0
        bool bullish = state.dir == 1
        // Capture the pre-raid pair while it is known, rather than selecting
        // the new P3 itself as P1 after its pivot becomes confirmed.
        float knownP1 = bullish ? lastPivotLow[1] : lastPivotHigh[1]
        float knownP2 = bullish ? lastPivotHigh[1] : lastPivotLow[1]
        int knownP1At = bullish ? lastPivotLowTime[1] : lastPivotHighTime[1]
        int knownP2At = bullish ? lastPivotHighTime[1] : lastPivotLowTime[1]
        bool orderedSeed = state.dir != 0 and not na(knownP1At) and not na(knownP2At) and knownP1At >= ref.c1Time and knownP1At < knownP2At and knownP2At < time and (bullish ? knownP1 < knownP2 : knownP1 > knownP2)
        bool seedRaid = orderedSeed and (bullish ? low <= knownP1 - sweepSize : high >= knownP1 + sweepSize)
        if seedRaid and (na(state.setupP1At) or na(state.reclaimedAt) and knownP1At > state.setupP1At)
            state.setupP1 := knownP1
            state.setupP1At := knownP1At
            state.setupP2 := knownP2
            state.setupP2At := knownP2At
        float target = bullish ? ref.c1H : ref.c1L
        bool targetTaken = state.dir != 0 and (bullish ? state.hi >= target : state.lo <= target)
        bool biasOk = biasDir == state.dir or (allowCrtCounter and biasDir == -state.dir and biasDir != 0)
        bool parentActive = qualifiedParent and parentDirection != 0 and not na(parentKnownAt) and parentKnownAt <= time
        if not state.complete
            state.status := "E DATA"
        else if not na(state.reclaimedAt) and not na(state.channel) and not na(state.channel.confirmIndex)
            bool stopped = bullish ? low <= state.attemptSl : high >= state.attemptSl
            bool reached = bullish ? high >= target : low <= target
            if not state.blocked and (stopped or reached)
                state.status := stopped and reached ? "E AMB" : stopped ? "E SL REF" : "E TP REF"
                state.terminalReason := state.status
                if reached
                    state.blocked := true
                else
                    // A stopped provisional trade stays stopped. Observation may
                    // still find a new raid before this HTF C2 has finished.
                    state.reclaimedAt := na
                    state.channel := ReactionState.new()
                    state.key := CrtKey.new()
                    state.status := "OBS NEW RAID"
            else if not state.blocked
                state.status := time_close >= ref.finish ? "C2 CLOSED" : state.emitted ? "E SENT" : "TRACK TS"
            // Observe later closed bars even after the one Early marker was sent.
            // No additional Early alert; confirmed parent can inherit this evidence.
            if not state.blocked and not na(state.channel) and not na(state.channel.confirmIndex)
                SignalContext tracking = SignalContext.new(c1T=ref.c1Time, c2T=ref.start, c2End=ref.finish, parentDir=parentDirection, parentAtKey=qualifiedParent, parentLive=qualifiedParent, parentConfirmed=parentKnownAt, age=0, crtKey=state.key, kind=1, id=ref.c1Time, dir=state.dir, confirmed=state.reclaimedAt, hi=ref.c1H, lo=ref.c1L, inv=state.attemptSl, cp=target)
                ChannelResult observed = f_stepChannel(tracking, state.channel, lastPivotHigh[1], lastPivotLow[1], na, na, enableEarlyC2 and permitEntry and not state.emitted and not confirmedHit and time_close < ref.finish, knownHighAt=lastPivotHighTime[1], knownLowAt=lastPivotLowTime[1])
                if observed.hit and not state.emitted
                    state.emitted := true
                    state.lockedSl := state.attemptSl
                    state.lockedTp := target
                    state.status := "E SIGNAL"
                    result.hit := true
                    result.dir := state.dir
                    result.inv := state.lockedSl
                    result.cp := state.lockedTp
                    result.confirmTime := observed.confirmTime
                    result.reclaimedAt := state.reclaimedAt
                    result.attempt := state.attempts
                    result.keyRef := observed.keyRef
        else if state.blocked
            state.status := state.terminalReason
        else if both or targetTaken
            state.blocked := true
            state.status := both ? "E DOUBLE" : "E TP TAKEN"
            state.terminalReason := state.status
        else if state.dir == 0
            state.status := "E WAIT RAID"
        else
            if extended or not inside
                state.reclaimedAt := na
                state.channel := ReactionState.new()
            CrtKey candidate = bullish ? buyKey : sellKey
            if na(state.key) or state.key.kind == 0
                if not na(candidate) and candidate.kind > 0 and candidate.dir == state.dir and not na(candidate.knownAt) and candidate.knownAt <= ref.start and candidate.beforeTime == ref.start
                    state.key := candidate.copy()
            bool hasKey = not na(state.key) and state.key.kind > 0 and state.key.dir == state.dir and state.key.knownAt <= ref.start
            bool keyHeld = hasKey and (bullish ? close > state.key.hi : close < state.key.lo)
            if hasKey and not keyHeld
                state.blocked := true
                state.status := "E KEY LOST"
                state.terminalReason := state.status
            else if not hasKey or not inside
                state.status := not hasKey ? "E NO KEY" : "E WAIT RECLAIM"
            else if na(state.reclaimedAt)
                state.attempts += 1
                state.reclaimedAt := time_close
                state.attemptSl := bullish ? state.lo : state.hi
                state.channel := ReactionState.new(sourceRoot=ref.c1Time, sourceC2=ref.start, sourceEnd=ref.finish, activatedAt=time_close, phaseAt=time_close)
                state.status := "E TS CLOSED"
                bool oppositeClose = bullish ? close > open : close < open
                if oppositeClose
                    state.channel.confirmIndex := bar_index
                    state.channel.confirmTime := time
                    state.channel.confirmLevel := bullish ? ref.c1L : ref.c1H
                    state.channel.localExtreme := state.attemptSl
                    state.channel.raidAt := ref.start
                    state.channel.p3At := bullish ? state.loAt : state.hiAt
                    state.channel.p3High := bullish ? state.lowCandleHigh : state.hi
                    state.channel.p3Low := bullish ? state.lo : state.highCandleLow
                    state.channel.p3Key := f_executionKey(state.dir, ref.start, state.channel.p3High, state.channel.p3Low, close)
                    state.channel.p3KeyAt := state.channel.p3Key.kind > 0 ? time_close : na
                    state.channel.legExtreme := bullish ? high : low
                    bool storedPair = not na(state.setupP1At) and not na(state.setupP2At)
                    float sourceP1 = storedPair ? state.setupP1 : knownP1
                    int sourceP1At = storedPair ? state.setupP1At : knownP1At
                    state.channel.mpsRef := storedPair ? state.setupP2 : not na(knownP2At) and knownP2At >= ref.c1Time ? knownP2 : na
                    state.channel.mpsRefAt := storedPair ? state.setupP2At : knownP2At
                    bool earlyPair = not na(state.channel.mpsRef) and not na(sourceP1At) and sourceP1At < state.channel.mpsRefAt and (bullish ? sourceP1 < state.channel.mpsRef : sourceP1 > state.channel.mpsRef)
                    bool earlyRaid = earlyPair and not na(state.channel.p3At) and state.channel.mpsRefAt < state.channel.p3At and state.channel.p3At <= time and (bullish ? state.lo <= sourceP1 - sweepSize : state.hi >= sourceP1 + sweepSize)
                    if earlyRaid
                        state.channel.p1Ref := sourceP1
                        state.channel.p1At := sourceP1At
                        state.channel.raidIndex := bar_index
                        state.channel.raidExtreme := state.attemptSl
                        state.channel.raidReclaimed := true
                    else
                        state.channel.mpsRef := na
                        state.channel.mpsRefAt := na
                    bool spentP2 = not na(state.channel.mpsRef) and (bullish ? structureBeforeClose > state.channel.mpsRef : structureBeforeClose < state.channel.mpsRef)
                    if spentP2
                        state.channel.mpsRef := na
                    bool initialMps = structureBarReady and state.channel.p3Key.kind > 0 and not na(state.channel.mpsRef) and (bullish ? structureClosing > state.channel.mpsRef and structurePriorClose <= state.channel.mpsRef : structureClosing < state.channel.mpsRef and structurePriorClose >= state.channel.mpsRef)
                    state.channel.mpsAt := initialMps ? structureClosedAt : na
                    state.channel.lastClosedAt := time_close
                state.channel.forwardKey := f_forwardKey(state.dir, time, close, target)
                state.channel.nextObjective := state.channel.forwardKey.kind > 0 ? (bullish ? state.channel.forwardKey.lo : state.channel.forwardKey.hi) : na
                state.channel.objectiveReached := false
                state.channel.objectiveAt := na
                bool freshEarlyKey = not na(state.channel.p3Key) and state.channel.p3Key.kind > 0 and f_executionProofValid(state.channel.p3Key, state.channel.p3KeyAt)
                if oppositeClose and not freshEarlyKey
                    state.status := "E WAIT P3 KEY"
                bool earlyCounter = biasDir == -state.dir or parentActive and parentDirection != state.dir
                bool earlyCounterSwing = not na(state.channel.p1At) and not na(state.channel.mpsRefAt) and not na(state.channel.p3At) and state.channel.p1At < state.channel.mpsRefAt and state.channel.mpsRefAt < state.channel.p3At and state.channel.p3At <= time and (bullish ? state.lo <= state.channel.p1Ref - sweepSize and close > state.channel.p1Ref : state.hi >= state.channel.p1Ref + sweepSize and close < state.channel.p1Ref)
                if oppositeClose and earlyCounter and not earlyCounterSwing
                    state.status := "E WAIT CT SWING"
                if oppositeClose and not state.emitted and (not earlyCounter or earlyCounterSwing) and (not parentActive or parentDirection == state.dir or allowCrtCounter and hasKey) and freshEarlyKey and enableEarlyC2 and permitEntry and biasOk and time_close < ref.finish and not confirmedHit and entryBranches != "FVG / OTE"
                    state.channel.branchMask := 1
                    state.channel.everSent := true
                    state.emitted := true
                    state.lockedSl := state.attemptSl
                    state.lockedTp := target
                    state.status := "E SIGNAL"
                    result.hit := true
                    result.dir := state.dir
                    result.inv := state.lockedSl
                    result.cp := state.lockedTp
                    result.confirmTime := time
                    result.reclaimedAt := state.reclaimedAt
                    result.attempt := state.attempts
                    result.keyRef := state.channel.p3Key
            else
                SignalContext local = SignalContext.new(age=0, crtKey=state.key, kind=1, id=ref.start, flags=1, phase=1, dir=state.dir, confirmed=state.reclaimedAt, resolved=state.reclaimedAt, hi=ref.c1H, lo=ref.c1L, inv=state.attemptSl, cp=target)
                ChannelResult entry = f_stepChannel(local, state.channel, lastPivotHigh[1], lastPivotLow[1], na, na, enableEarlyC2 and permitEntry and not confirmedHit and time_close < ref.finish, knownHighAt=lastPivotHighTime[1], knownLowAt=lastPivotLowTime[1])
                state.status := confirmedHit ? "E C3 PRIORITY" : "E " + entry.status


                if entry.hit and not confirmedHit
                    state.emitted := true
                    state.lockedSl := state.attemptSl
                    state.lockedTp := target
                    state.status := "E SIGNAL"
                    result.hit := true
                    result.dir := state.dir
                    result.inv := state.lockedSl
                    result.cp := state.lockedTp
                    result.confirmTime := entry.confirmTime
                    result.reclaimedAt := state.reclaimedAt
                    result.attempt := state.attempts
                    result.keyRef := entry.keyRef
    result.status := not isLtf ? "E VIEW TF" : not enableEarlyC2 or not permitEntry ? "OBS | " + state.status : state.status
    result

type ClusterCandle
    int t
    float o
    float h
    float l
    float c
// Merge frozen source candles only where chart-derived OHLC is unavailable.
// Keep chronological order, including all intervening CRT follow candles.
f_insertCluster(array<ClusterCandle> candles, int t, float o, float h, float l, float c) =>
    if not na(t) and not na(o)
        int insertion = array.size(candles)
        bool exists = false
        if array.size(candles) > 0
            for j = 0 to array.size(candles) - 1
                int itemTime = array.get(candles, j).t
                if itemTime == t
                    exists := true
                    break
                if itemTime > t
                    insertion := j
                    break
        if not exists
            array.insert(candles, insertion, ClusterCandle.new(t, o, h, l, c))
// MODEL and FOLLOW are disjoint slices of the same chronological candle series.
f_splitCluster(array<ClusterCandle> seriesCandles, int kind, int aT, int bT, int cT, int followLimit, array<ClusterCandle> origin, array<ClusterCandle> follow) =>
    array.clear(origin)
    array.clear(follow)
    int sourceEnd = bT
    int followTotal = 0
    if array.size(seriesCandles) > 0
        for j = 0 to array.size(seriesCandles) - 1
            ClusterCandle item = array.get(seriesCandles, j)
            bool isOrigin = item.t == aT or item.t == bT
            if isOrigin
                array.push(origin, item)
            else if item.t > sourceEnd
                array.push(follow, item)
                followTotal += 1
    while array.size(follow) > followLimit
        array.shift(follow)
    followTotal

f_archiveCandles(ModelArchive ended, array<ClusterCandle> candles) =>
    array.clear(candles)
    if not na(ended) and not na(ended.uid)
        f_insertCluster(candles, ended.aT, ended.aO, ended.aH, ended.aL, ended.aC)
        f_insertCluster(candles, ended.bT, ended.bO, ended.bH, ended.bL, ended.bC)
        if ended.kind == 1
            f_insertCluster(candles, ended.cT, ended.cO, ended.cH, ended.cL, ended.cC)
