f_patterns(float c1o, float c1h, float c1l, float c1c, float c2o, float c2h, float c2l, float c2c) =>
    float lowRef = tsReference == "C1 range" ? c1l : manualSsl > 0 ? manualSsl : na
    float highRef = tsReference == "C1 range" ? c1h : manualBsl > 0 ? manualBsl : na
    bool tookLow = not na(lowRef) and c2l <= lowRef - sweepSize
    bool tookHigh = not na(highRef) and c2h >= highRef + sweepSize
    bool reclaimed = c2c > c1l and c2c < c1h
    bool sweptBothC1 = c2l <= c1l - sweepSize and c2h >= c1h + sweepSize
    bool doubleSweep = sweptBothC1 or (tookLow and tookHigh)
    bool tsBull = tookLow and not doubleSweep and reclaimed and c2c > lowRef
    bool tsBear = tookHigh and not doubleSweep and reclaimed and c2c < highRef
    bool outsideBar = c2h > c1h and c2l < c1l
    bool insideBar = c2h <= c1h and c2l >= c1l and (c2h < c1h or c2l > c1l)
    float prevBodyHigh = math.max(c1o, c1c)
    float prevBodyLow = math.min(c1o, c1c)
    float bodyHigh = math.max(c2o, c2c)
    float bodyLow = math.min(c2o, c2c)
    bool bodyCovers = bodyHigh >= prevBodyHigh and bodyLow <= prevBodyLow and (bodyHigh > prevBodyHigh or bodyLow < prevBodyLow)
    bool engulfBull = c2c > c2o and c1c < c1o and bodyCovers
    bool engulfBear = c2c < c2o and c1c > c1o and bodyCovers
    [tsBull, tsBear, outsideBar, insideBar, engulfBull, engulfBear]

f_biasCore() =>
    var int direction = 0
    var int age = 0
    bool lowTaken = low < low[1]
    bool highTaken = high > high[1]
    bool inPrior = close > low[1] and close < high[1]
    int nextDir = lowTaken and not highTaken and inPrior ? 1 : highTaken and not lowTaken and inPrior ? -1 : close > high[1] ? 1 : close < low[1] ? -1 : 0
    age += 1
    if nextDir != 0
        direction := nextDir
        age := 0
    else if (lowTaken and highTaken and inPrior) or age >= biasLife
        direction := 0
    [direction, age]
f_biasSnapshot(bool previous) =>
    [direction, age] = f_biasCore()
    [previous ? direction[1] : direction, previous ? age[1] : age]

f_refreshIrl(Model m, array<Zone> zones, array<Liquidity> swings) =>
    m.bullIrlLeft := na
    m.bearIrlLeft := na
    m.bullIrlH := na
    m.bullIrlL := na
    m.bullIrlT := na
    m.bullIrlKind := 0
    m.bearIrlH := na
    m.bearIrlL := na
    m.bearIrlT := na
    m.bearIrlKind := 0
    m.innerHi := na
    m.innerLo := na
    m.innerHiT := na
    m.innerLoT := na
    // Separate candidate maps: do not discard the other direction just because bias is set.
    for pass = 1 to 2
        if array.size(zones) > 0
            for offset = 0 to array.size(zones) - 1
                Zone z = array.get(zones, array.size(zones) - 1 - offset)
                bool internalZone = z.kind == pass and z.hi <= m.hi and z.lo >= m.lo
                if internalZone and z.dir == 1 and na(m.bullIrlT)
                    m.bullIrlH := z.hi
                    m.bullIrlL := z.lo
                    m.bullIrlT := z.born
                    m.bullIrlKind := z.kind
                    m.bullIrlLeft := z.leftTime
                if internalZone and z.dir == -1 and na(m.bearIrlT)
                    m.bearIrlH := z.hi
                    m.bearIrlL := z.lo
                    m.bearIrlT := z.born
                    m.bearIrlKind := z.kind
                    m.bearIrlLeft := z.leftTime
    if array.size(swings) > 0
        for j = 0 to array.size(swings) - 1
            Liquidity q = array.get(swings, j)
            bool internalSwing = q.price < m.hi and q.price > m.lo
            if internalSwing and q.side == 1 and (na(m.innerHi) or q.price > m.innerHi)
                m.innerHi := q.price
                m.innerHiT := q.born
            if internalSwing and q.side == -1 and (na(m.innerLo) or q.price < m.innerLo)
                m.innerLo := q.price
                m.innerLoT := q.born
    if na(m.irlT) and m.dir != 0
        bool bull = m.dir == 1
        m.irlH := bull ? m.bullIrlH : m.bearIrlH
        m.irlL := bull ? m.bullIrlL : m.bearIrlL
        m.irlT := bull ? m.bullIrlT : m.bearIrlT
        m.irlLeft := bull ? m.bullIrlLeft : m.bearIrlLeft
        m.irlKind := bull ? m.bullIrlKind : m.bearIrlKind
        m.irlDir := m.dir
        // EGF can react at a genuine internal SSL/BSL point when no zone exists.
        if m.kind == 2 and na(m.irlT)
            m.irlH := bull ? m.innerLo : m.innerHi
            m.irlL := m.irlH
            m.irlT := bull ? m.innerLoT : m.innerHiT
            m.irlLeft := m.irlT
            m.irlKind := not na(m.irlT) ? 3 : 0
    // Preserve the selected reaction zone through wick mitigation until HTF close-through.
    if not na(m.irlT) and m.irlKind != 3
        if m.irlDir == 1
            m.bullIrlH := m.irlH
            m.bullIrlL := m.irlL
            m.bullIrlT := m.irlT
            m.bullIrlKind := m.irlKind
            m.bullIrlLeft := m.irlLeft
        else
            m.bearIrlH := m.irlH
            m.bearIrlL := m.irlL
            m.bearIrlT := m.irlT
            m.bearIrlKind := m.irlKind
            m.bearIrlLeft := m.irlLeft
    if not na(m.irlT) and m.irlKind == 3
        if m.irlDir == 1
            m.innerLo := m.irlL
            m.innerLoT := m.irlT
        else
            m.innerHi := m.irlH
            m.innerHiT := m.irlT

f_engine() =>
    [localBias, localBiasAge] = f_biasCore()
    [higherBias, higherBiasAge] = request.security(syminfo.tickerid, biasTf, f_biasSnapshot(true), lookahead=barmerge.lookahead_on, calc_bars_count=5000)
    int contextBias = biasMode == "Auto" ? (biasTf == htf ? localBias : nz(higherBias)) : biasMode == "Long" ? 1 : biasMode == "Short" ? -1 : 0
    var array<Model> models = array.new<Model>()
    var array<Zone> zones = array.new<Zone>()
    var array<Liquidity> swings = array.new<Liquidity>()
    var int lastRetired = na
    var int retiredReason = 0
    var int dropped = 0
    var int firstSeen = time
    int scanStart = last_bar_time - int(timeframe.in_seconds() * 1000) * scanDepth
    bool inScan = time >= scanStart
    float ph = ta.pivothigh(high, htfPivotLength, htfPivotLength)
    float pl = ta.pivotlow(low, htfPivotLength, htfPivotLength)
    // Remove consumed or out-of-window references, before looking for new CP levels.
    int si = array.size(swings) - 1
    while si >= 0
        Liquidity q = array.get(swings, si)
        if q.born < scanStart or (q.side == 1 ? high >= q.price : low <= q.price)
            array.remove(swings, si)
        si -= 1
    if inScan and not na(ph) and time[htfPivotLength] >= scanStart and high < ph
        array.push(swings, Liquidity.new(ph, 1, time[htfPivotLength]))
    if inScan and not na(pl) and time[htfPivotLength] >= scanStart and low > pl
        array.push(swings, Liquidity.new(pl, -1, time[htfPivotLength]))
    if array.size(swings) > 80
        array.shift(swings)
    int zi = array.size(zones) - 1
    while zi >= 0
        Zone z = array.get(zones, zi)
        bool filled = z.dir == 1 ? low <= z.lo : high >= z.hi
        if z.born < scanStart or filled
            array.remove(zones, zi)
        zi -= 1
    if inScan and not na(high[2])
        if low > high[2]
            array.push(zones, Zone.new(leftTime=time[2], hi=low, lo=high[2], dir=1, born=time, kind=1))
        if high < low[2]
            array.push(zones, Zone.new(leftTime=time[2], hi=low[2], lo=high, dir=-1, born=time, kind=1))
        if close[1] < open[1] and close > high[1]
            array.push(zones, Zone.new(leftTime=time[1], hi=open[1], lo=low[1], dir=1, born=time[1], kind=2))
        if close[1] > open[1] and close < low[1]
            array.push(zones, Zone.new(leftTime=time[1], hi=high[1], lo=open[1], dir=-1, born=time[1], kind=2))
    while array.size(zones) > 60
        array.shift(zones)

    // Retire/resolve existing models first. Newly observed models never displace older valid ones.
    int mi = array.size(models) - 1
    while mi >= 0
        Model m = array.get(models, mi)
        if na(m.cT) and time > m.bT and (m.dir != 0 or m.kind == 2)
            m.cO := open
            m.cH := high
            m.cL := low
            m.cC := close
            m.cT := time
        // EGF direction is a bias-led scenario, never the color of the engulfing candle.
        if m.kind == 2 and m.dir != contextBias
            m.dir := contextBias
            m.phase := 0
            m.resolvedAt := time_close
            m.resolvedIndex := bar_index
            m.inv := contextBias == 1 ? m.lo : contextBias == -1 ? m.hi : na
            m.invT := contextBias == 1 ? m.loT : contextBias == -1 ? m.hiT : na
            m.cp := na
            m.cpT := na
            m.irlH := na
            m.irlL := na
            m.irlT := na
            m.irlKind := 0
        bool older = time_close > m.confirmedAt
        bool bad = older and m.dir != 0 and (m.dir == 1 ? (invalidMode == "Close" ? close < m.inv : low < m.inv) : (invalidMode == "Close" ? close > m.inv : high > m.inv))
        bool reached = older and not na(m.cp) and m.dir != 0 and (m.dir == 1 ? high >= m.cp : low <= m.cp)
        bool expired = (lifeHtf > 0 and bar_index - m.bornIndex >= lifeHtf) or m.uid < scanStart
        if bad or reached or expired
            lastRetired := m.uid
            retiredReason := bad and reached ? 4 : bad ? 1 : reached ? 2 : 3
            array.remove(models, mi)
        else
            if m.kind == 3 and m.dir == 0 and older
                bool sweepLo = low <= m.lo - sweepSize
                bool sweepHi = high >= m.hi + sweepSize
                bool backIn = close > m.lo and close < m.hi
                int resolveDir = sweepLo and not sweepHi and backIn ? 1 : sweepHi and not sweepLo and backIn ? -1 : close > m.hi ? 1 : close < m.lo ? -1 : 0
                if sweepLo and sweepHi and backIn
                    lastRetired := m.uid
                    retiredReason := 4
                    array.remove(models, mi)
                else if resolveDir != 0
                    m.dir := resolveDir
                    m.phase := backIn ? 1 : 2
                    m.resolvedAt := time_close
                    m.resolvedIndex := bar_index

    bool egfScenario = sKind == 2
    result.status := not isLtf ? "VIEW TF" : not armed ? (na(ctx.id) ? "NO MODEL" : "NO DIR") : "WAIT CLOSE"
    if armed and barstate.isconfirmed and isLtf and time >= activatedAt
        bool bullish = ctx.dir == 1
        bool opposing = bullish ? close < open : close > open
        bool priorOpposing = bullish ? close[1] < open[1] : close[1] > open[1]
        float seedPivot = bullish ? knownHigh : knownLow
        bool changedIrl = not na(ctx.irlT) ? na(state.trackedIrl) or state.trackedIrl != ctx.irlT : not na(state.trackedIrl)
        if changedIrl

            if egfScenario
                f_clearReaction(state, na, seedPivot)
        bool touchesIrl = not na(sIrlH) and not na(sIrlL) and low <= sIrlH and high >= sIrlL
        bool outsideIrl = not na(sIrlH) and not na(sIrlL) and (bullish ? low > sIrlH : high < sIrlL)
        if state.sent and outsideIrl
